create table id_records (
  id uuid default gen_random_uuid() primary key,
  companyId text,
  fullName text,
  contactNumber text,
  email text,
  emergencyName text,
  emergencyContact text,
  philHealth text,
  sss text,
  tin text,
  pagIbig text,
  birthday date,
  created_at timestamptz default now()
);

-- This allows the website to save data without needing a login
alter table id_records enable row level security;
create policy "Allow public insert" on id_records for insert with check (true);
